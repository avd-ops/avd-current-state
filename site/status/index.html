Run this standalone block in Azure Cloud Shell (PowerShell) against your avd-sandbox subscription ($0 spend, read-only). Download the emitted avd-current-state-facts-*.json and avd-assessment-*.json files and drop them onto Tab 1:

if (-not (Test-Path ~/avd-current-state)) { git clone https://github.com/avd-ops/avd-current-state.git ~/avd-current-state }
Set-Location ~/avd-current-state; git checkout -q main; git pull -q --ff-only
$out = Join-Path $HOME "clouddrive/avd-status-run"
New-Item -ItemType Directory -Path $out -Force | Out-Null
./avd_current_state_collector.ps1 -OutputDirectory (Join-Path $out "pkg")
$pkg = Get-ChildItem (Join-Path $out "pkg") -Directory | Sort-Object LastWriteTime -Descending | Select-Object -First 1
./avd_current_state_facts.ps1 -PackagePath $pkg.FullName -KeyPath (Join-Path $out "estate.key") -NewKey -OutputPath $out -Assess -StatusReport
